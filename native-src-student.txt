class _CodeEntryPageState extends State<CodeEntryPage> {
  final _code = TextEditingController();
  bool _busy = false;

  Future<void> _redeem() async {
    final token = _code.text.trim().toUpperCase();
    if (token.isEmpty) { toast(context, 'اكتب الكود الأول', err: true); return; }
    setState(() => _busy = true);
    try {
      final r = await sb.rpc('redeem_code', params: {'p_token': token});
      if (!mounted) return;
      if (r['ok'] != true) {
        final m = {
          'invalid_code': 'الكود ده مش مظبوط. اتأكد منه وكلم مدرسك.',
          'not_published': 'الامتحان ده لسه مش متاح.',
          'not_yet': 'الامتحان لسه ما بدأش. استنى المعاد اللي قال عليه مدرسك.',
          'expired': 'وقت الامتحان ده خلص. كلم مدرسك.',
          'already_completed': 'الكود ده اتستخدم وخلاص. كلم مدرسك لو في غلط.',
        };
        toast(context, m[r['error']] ?? 'حصلت مشكلة، جرب تاني', err: true);
        setState(() => _busy = false);
        return;
      }
      Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => ExamIntroPage(data: r)));
    } catch (_) {
      if (mounted) { toast(context, 'حصلت مشكلة في الاتصال، جرب تاني', err: true); setState(() => _busy = false); }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(title: const Text('دخول الامتحان')),
        body: ListView(padding: const EdgeInsets.all(20), children: [
          Card(
            child: Padding(
              padding: const EdgeInsets.all(20),
              child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
                const Text('بوابة الامتحانات', style: TextStyle(fontSize: 19, fontWeight: FontWeight.w800), textAlign: TextAlign.center),
                const SizedBox(height: 6),
                Text('اكتب الكود اللي المدرس بعتهولك', style: TextStyle(color: Colors.grey.shade600), textAlign: TextAlign.center),
                const SizedBox(height: 16),
                TextField(

class TakeExamPage extends StatefulWidget {
  final Map data;
  const TakeExamPage({super.key, required this.data});
  @override
  State<TakeExamPage> createState() => _TakeExamPageState();
}

class _TakeExamPageState extends State<TakeExamPage> with WidgetsBindingObserver {
  late List _questions;
  late Map _answers; // question_id -> dynamic (int index for mcq/tf, string for sa/essay)
  late DateTime _startedAt;
  late int _durationSec;
  Timer? _timer;
  int _remaining = 0;
  int _index = 0;
  bool _submitting = false;

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    final exam = widget.data['exam'] as Map;
    _questions = List.from(widget.data['questions'] ?? []);
    if (exam['shuffle_q'] == true) _questions.shuffle(Random());
    // choices carry their original index so shuffled display still grades right
    for (final q in _questions) {
      if (q['type'] == 'mcq' && q['choices'] is List) {
        final mapped = [for (var i = 0; i < (q['choices'] as List).length; i++) {'t': q['choices'][i], 'i': i}];
        if (exam['shuffle_c'] == true) mapped.shuffle(Random());
        q['choices'] = mapped;
      } else if (q['type'] == 'tf') {
        q['choices'] = [
          {'t': (q['choices'] as List)[0], 'i': 0},
          {'t': (q['choices'] as List)[1], 'i': 1},
        ];
      }
    }
    _answers = Map.from(widget.data['saved'] ?? {});
    _startedAt = DateTime.tryParse(widget.data['exam']['started_at'] ?? '') ?? DateTime.now();
    _durationSec = (widget.data['exam']['duration_min'] ?? 15) * 60;
    final elapsed = DateTime.now().difference(_startedAt.toLocal()).inSeconds;
    _remaining = (_durationSec - elapsed).clamp(0, _durationSec);
    _timer = Timer.periodic(const Duration(seconds: 1), (t) {
      if (_remaining <= 1) { t.cancel(); _submit(auto: true); return; }
      if (mounted) setState(() => _remaining--);
    });
  }

  final Map<dynamic, TextEditingController> _ctrls = {};
  final Map<dynamic, Timer> _debounce = {};
  Timer? _retry;

  @override
  void dispose() {
    _timer?.cancel(); _retry?.cancel();
    for (final t in _debounce.values) { t.cancel(); }
    for (final c in _ctrls.values) { c.dispose(); }
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  void _flushText() {
    final pending = _debounce.keys.toList();
    for (final k in pending) { _debounce.remove(k)?.cancel(); _save(k, _answers[k]); }
  }

  String _saveNote = '';
  bool _expired = false;

  void _save(dynamic qid, dynamic answer) {
    _answers[qid] = answer;
    if (mounted) setState(() => _saveNote = 'جاري الحفظ…');
    sb.rpc('save_answer', params: {'p_session': widget.data['session_id'], 'p_question': qid, 'p_answer': answer}).then((r) {
      if (!mounted) return;
      if (r is Map && (r['rejected'] == 'time_expired' || r['error'] == 'time_expired')) {
        // server says time is over: stop editing and hand in what was saved
        if (!_expired) { _expired = true; _submit(auto: true); }
        return;
      }
      setState(() => _saveNote = (r is Map && r['ok'] == false) ? 'مشكلة في الحفظ، اتأكد من النت' : '✓ اتحفظت');
    }).catchError((_) { if (mounted) setState(() => _saveNote = 'مشكلة في الحفظ، اتأكد من النت'); });
  }

  Future<void> _submit({bool auto = false}) async {
    if (_submitting) return;
    _flushText();
    if (!auto) {
      final yes = await showDialog<bool>(
        context: context,
        builder: (ctx) => Directionality(
          textDirection: TextDirection.rtl,
          child: AlertDialog(
            title: const Text('تسليم الامتحان؟'),
            content: const Text('متأكد إنك عايز تسلّم؟ مش هتقدر ترجع بعدها.'),
            actions: [
              TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('لسه')),
              ElevatedButton(onPressed: () => Navigator.pop(ctx, true), child: const Text('سلّم')),
            ],
          ),
        ),
      );
      if (yes != true) return;
    }
    setState(() => _submitting = true);
    try {
      final timeSec = DateTime.now().difference(_startedAt.toLocal()).inSeconds;
      final r = await sb.rpc('submit_exam', params: {
        'p_session': widget.data['session_id'], 'p_time_sec': timeSec, 'p_auto': auto,
      });
      if (!mounted) return;
      Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => ExamResultPage(result: r, student: widget.data['student'] ?? '')));
    } catch (_) {
      if (mounted) {
        setState(() { _submitting = false; _saveNote = auto ? 'الوقت خلص. مفيش نت، هنحاول نسلّم تاني لحد ما ينجح…' : ''; });
        if (auto) { _retry?.cancel(); _retry = Timer(const Duration(seconds: 8), () { if (mounted) _submit(auto: true); }); }
        else { toast(context, 'حصلت مشكلة في التسليم، جرب تاني', err: true); }
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final q = _questions[_index];
    final mins = (_remaining ~/ 60).toString().padLeft(2, '0');
    final secs = (_remaining % 60).toString().padLeft(2, '0');
    final low = _remaining < 60;
    return Directionality(
      textDirection: TextDirection.rtl,
      child: PopScope(
        canPop: false,
        onPopInvokedWithResult: (didPop, _) async {
          if (didPop) return;
          final leave = await showDialog<bool>(
            context: context,
            builder: (ctx) => Directionality(
              textDirection: TextDirection.rtl,
              child: AlertDialog(
                title: const Text('تخرج من الامتحان؟'),
                content: const Text('الوقت هيفضل شغال، وإجاباتك اتحفظت. تقدر ترجع بنفس الكود طول ما الوقت لسه موجود.'),
                actions: [
                  TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('كمّل امتحان')),
                  TextButton(onPressed: () => Navigator.pop(ctx, true), child: const Text('اخرج')),
                ],
              ),
            ),
          );
          if (leave == true && context.mounted) { _flushText(); Navigator.of(context).popUntil((r) => r.isFirst); }
        },
        child: Scaffold(
          appBar: AppBar(
            automaticallyImplyLeading: false,
            title: Text('سؤال ${_index + 1} من ${_questions.length}'),
            actions: [
              Center(child: Container(
                margin: const EdgeInsetsDirectional.only(end: 14),
                padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 5),
                decoration: BoxDecoration(
                  color: low ? kTeal : (_remaining < 300 ? kGold : Colors.white),
                  borderRadius: BorderRadius.circular(20),
                  border: Border.all(color: kNavy, width: 2),
                ),
                child: Text('$mins:$secs', style: TextStyle(fontWeight: FontWeight.w900, fontSize: 16, color: low ? Colors.white : kNavy)),
              )),
            ],
          ),
          body: Column(children: [
            LinearProgressIndicator(minHeight: 7, value: (_index + 1) / _questions.length, color: kTeal, backgroundColor: const Color(0xFFEADFC9)),
            Expanded(
              child: ListView(padding: const EdgeInsets.all(20), children: [
                Card(
                  child: Padding(
                    padding: const EdgeInsets.all(20),
                    child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
                      Text(q['text'] ?? '', style: const TextStyle(fontSize: 18, fontWeight: FontWeight.w800)),
                      const SizedBox(height: 4),
                      Text('${q['points']} درجة', style: TextStyle(color: Colors.grey.shade500, fontSize: 12)),
                      const SizedBox(height: 16),
                      _answerWidget(q),
                    ]),
                  ),
                ),
              ]),
            ),
            if (_saveNote.isNotEmpty)
              Padding(padding: const EdgeInsets.only(top: 6), child: Text(_saveNote, style: TextStyle(fontSize: 12, fontWeight: FontWeight.w700, color: _saveNote.startsWith('✓') ? const Color(0xFF2F8F5B) : (_saveNote.startsWith('مشكلة') ? Colors.red.shade700 : Colors.grey.shade600)))),
            Padding(
              padding: const EdgeInsets.all(16),
              child: Row(children: [
                if (_index > 0)
                  Expanded(child: OutlinedButton(onPressed: () => setState(() => _index--), child: const Text('السابق'))),
                if (_index > 0) const SizedBox(width: 10),
                Expanded(
                  flex: 2,
                  child: _index < _questions.length - 1
                      ? ElevatedButton(onPressed: () => setState(() => _index++), child: const Text('اللي بعده'))
                      : BusyBtn(label: 'تسليم الامتحان', busy: _submitting, onTap: () => _submit(), icon: Icons.check),
                ),
              ]),
            ),
          ]),
        ),
      ),
    );
  }

  Widget _answerWidget(Map q) {
    final qid = q['id'];
    final type = q['type'];
    if (type == 'mcq' || type == 'tf') {
      final choices = (q['choices'] as List?) ?? [];
      final sel = _answers[qid];
      return Column(
        children: List.generate(choices.length, (row) {
          final origIdx = choices[row]['i'];
          final isSel = sel == origIdx || (sel is String && sel == '$origIdx');
          return Padding(
            padding: const EdgeInsets.only(bottom: 8),
            child: InkWell(
              borderRadius: BorderRadius.circular(14),
              onTap: () => setState(() => _save(qid, origIdx)),
              child: Container(
                padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 14),
                decoration: BoxDecoration(
                  color: isSel ? kMint : Colors.white,
                  borderRadius: BorderRadius.circular(18),
                  border: Border.all(color: kNavy, width: 2),
                  boxShadow: isSel ? const [BoxShadow(color: kNavy, offset: Offset(3, 3))] : null,
                ),
                child: Row(children: [
                  Container(
                    width: 32, height: 32, alignment: Alignment.center,
                    decoration: BoxDecoration(color: isSel ? kNavy : kBg, borderRadius: BorderRadius.circular(10), border: Border.all(color: kNavy, width: 2)),
                    child: Text(row < 4 ? 'أبجد'[row] : '${row + 1}', style: TextStyle(fontWeight: FontWeight.w900, fontSize: 14, color: isSel ? Colors.white : kNavy)),
                  ),
                  const SizedBox(width: 12),
                  Expanded(child: Text('${choices[row]['t']}', style: const TextStyle(color: kNavy, fontWeight: FontWeight.w700, fontSize: 15))),
                ]),
              ),
            ),
          );
        }),
      );
    }
    // sa / essay: free text
    return TextField(
      maxLines: type == 'essay' ? 6 : 2,
      controller: _ctrls.putIfAbsent(qid, () => TextEditingController(text: _answers[qid]?.toString() ?? '')),
      decoration: InputDecoration(hintText: type == 'essay' ? 'اكتب إجابتك بالتفصيل...' : 'اكتب إجابتك...'),
      onChanged: (v) {
        _answers[qid] = v;
        _debounce[qid]?.cancel();
        _debounce[qid] = Timer(const Duration(milliseconds: 700), () { _debounce.remove(qid); _save(qid, v); });
      },
    );
  }
}

import 'package:flutter/material.dart';
import '../src/core.dart';

class FaqBotPage extends StatefulWidget {
  const FaqBotPage({super.key});
  @override
  State<FaqBotPage> createState() => _FaqBotPageState();
}

class _Msg {
  final String text;
  final bool fromUser;
  const _Msg(this.text, this.fromUser);
}

class _FaqBotPageState extends State<FaqBotPage> {
  final _input = TextEditingController();
  final _scroll = ScrollController();
  final List<_Msg> _msgs = [];
  List _faqs = [];
  bool _loading = true;

  @override
  void initState() {
    super.initState();
    _load();
  }

  Future<void> _load() async {
    try {
      final r = await sb.rpc('faq_list');
      if (mounted) {
        setState(() { _faqs = r is List ? r : []; _loading = false; });
        _msgs.add(_Msg(_faqs.isEmpty
            ? 'أهلًا بيك! 👋 أنا بوت الأسئلة المتكررة. لسه مفيش أسئلة اتعلمتها، أول ما المدرس يضيف أسئلة وأجوبة هقدر أرد عليك.'
            : 'أهلًا بيك! 👋 اسألني أي سؤال من الأسئلة المتكررة وهجاوبك على طول.', false));
      }
    } catch (_) {
      if (mounted) setState(() => _loading = false);
    }
  }

  String _norm(String s) => s
      .replaceAll(RegExp(r'[أإآ]'), 'ا')
      .replaceAll('ة', 'ه')
      .replaceAll('ى', 'ي')
      .replaceAll(RegExp(r'[^\u0621-\u064Aa-zA-Z0-9\s]'), ' ')
      .replaceAll(RegExp(r'\s+'), ' ')
      .trim()
      .toLowerCase();

  void _ask() {
    final q = _input.text.trim();
    if (q.isEmpty) return;
    _input.clear();
    setState(() => _msgs.add(_Msg(q, true)));
    final nq = _norm(q);
    final qWords = nq.split(' ').where((w) => w.length > 2).toSet();
    Map? best;
    var bestScore = 0;
    for (final f in _faqs) {
      final fq = _norm(f['question'] ?? '');
      final fWords = fq.split(' ').where((w) => w.length > 2).toSet();
      final common = qWords.intersection(fWords).length;
      final contains = fq.contains(nq) || nq.contains(fq);
      final score = common * 2 + (contains ? 5 : 0);
      if (score > bestScore) { bestScore = score; best = f; }
    }
    final answer = (best != null && bestScore >= 4)
        ? (best['answer'] ?? '').toString()
        : (_faqs.isEmpty
            ? 'لسه مفيش إجابات متسجلة. اسأل المدرس مباشرة دلوقتي.'
            : 'مش لاقي إجابة للسؤال ده عندي. جرب تسأل بطريقة تانية، أو اسأل المدرس مباشرة.');
    Future.delayed(const Duration(milliseconds: 500), () {
      if (mounted) {
        setState(() => _msgs.add(_Msg(answer, false)));
        _scroll.animateTo(_scroll.position.maxScrollExtent + 120, duration: const Duration(milliseconds: 300), curve: Curves.easeOut);
      }
    });
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(title: const Row(mainAxisSize: MainAxisSize.min, children: [Icon(Icons.smart_toy_outlined, size: 22), SizedBox(width: 6), Text('بوت الأسئلة')])),
        body: Column(children: [
          Expanded(
            child: _loading
                ? const Center(child: CircularProgressIndicator())
                : ListView.builder(
                    controller: _scroll,
                    padding: const EdgeInsets.all(16),
                    itemCount: _msgs.length,
                    itemBuilder: (_, i) {
                      final m = _msgs[i];
                      return Align(
                        alignment: m.fromUser ? AlignmentDirectional.centerStart : AlignmentDirectional.centerEnd,
                        child: Container(
                          margin: const EdgeInsets.symmetric(vertical: 4),
                          padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 10),
                          constraints: BoxConstraints(maxWidth: MediaQuery.of(context).size.width * 0.8),
                          decoration: BoxDecoration(

class _VideoLoginPageState extends State<VideoLoginPage> {
  final _user = TextEditingController();
  final _pass = TextEditingController();
  bool _busy = false;

  @override
  void initState() {
    super.initState();
    _trySaved();
  }

  Future<void> _trySaved() async {
    final prefs = await SharedPreferences.getInstance();
    const ss = FlutterSecureStorage();
    String? u, p;
    try {
      // one-time migration: move any old plaintext credentials into the keystore, then delete them
      final ou = prefs.getString('vuser'), op = prefs.getString('vpass');
      if (ou != null && op != null) {
        await ss.write(key: 'vuser', value: ou);
        await ss.write(key: 'vpass', value: op);
      }
      await prefs.remove('vuser'); await prefs.remove('vpass');
      u = await ss.read(key: 'vuser'); p = await ss.read(key: 'vpass');
    } catch (_) {
      await prefs.remove('vuser'); await prefs.remove('vpass');
    }
    if (u != null && p != null && u.isNotEmpty) _login(u, p, quiet: true);
  }

  Future<void> _login(String u, String p, {bool quiet = false}) async {
    if (u.trim().isEmpty || p.isEmpty) { if (!quiet) toast(context, 'اكتب اسم المستخدم وكلمة السر', err: true); return; }
    setState(() => _busy = true);
    try {
      final r = await sb.rpc('video_login', params: {'p_username': u.trim(), 'p_password': p});
      if (!mounted) return;
      if (r['ok'] != true) {
        final m = {'invalid': 'اليوزر أو الباسورد مش مظبوطين.', 'inactive': 'الاشتراك ده متوقف حاليًا. كلم مدرسك.', 'throttled': 'محاولات كتير غلط. استنى ١٠ دقايق وجرّب تاني.'};
        if (!quiet) toast(context, m[r['error']] ?? 'حصلت مشكلة، جرب تاني', err: true);
        setState(() => _busy = false);
        return;
      }
      try {
        const ss = FlutterSecureStorage();
        await ss.write(key: 'vuser', value: u.trim());
        await ss.write(key: 'vpass', value: p);
      } catch (_) {}
      if (!mounted) return;
      Navigator.pushReplacement(context, MaterialPageRoute(builder: (_) => VideoLibraryPage(name: r['name'] ?? '', videos: r['videos'] ?? [], stage: widget.stage)));
    } catch (_) {
      if (mounted && !quiet) { toast(context, 'حصلت مشكلة في الاتصال، جرب تاني', err: true); setState(() => _busy = false); }
      else if (mounted) setState(() => _busy = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(title: const Text('فيديوهات الشرح')),
        body: ListView(padding: const EdgeInsets.all(20), children: [
          FadeSlideIn(
            child: Card(
              child: Padding(
                padding: const EdgeInsets.all(20),
                child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
                  const Text('دخول المشتركين', style: TextStyle(fontSize: 19, fontWeight: FontWeight.w800), textAlign: TextAlign.center),
                  const SizedBox(height: 6),
                  Text('ادخل باليوزر والباسورد اللي المدرس دهوملك', style: TextStyle(color: Colors.grey.shade600), textAlign: TextAlign.center),
                  const SizedBox(height: 16),
                  TextField(
                    controller: _user, textDirection: TextDirection.ltr, textAlign: TextAlign.left,
                    decoration: const InputDecoration(labelText: 'اسم المستخدم', prefixIcon: Icon(Icons.person_outline)),
                  ),
                  const SizedBox(height: 12),
                  PwField(controller: _pass, action: TextInputAction.done, onSubmit: (_) => _login(_user.text, _pass.text)),
                  const SizedBox(height: 16),
                  BusyBtn(label: 'دخول للفيديوهات', busy: _busy, onTap: () => _login(_user.text, _pass.text), icon: Icons.play_arrow),
                ]),
              ),
            ),
          ),
          const SizedBox(height: 12),
          Card(
            child: ListTile(
              leading: const Icon(Icons.payments_outlined, color: kGold),
              title: const Text('لسه مش مشترك؟', style: TextStyle(fontWeight: FontWeight.w700)),
              subtitle: const Text('دوس هنا عشان تدفع فودافون كاش'),
              onTap: () => showVodafonePaySheet(context),
            ),
          ),
        ]),
      ),
    );
  }
