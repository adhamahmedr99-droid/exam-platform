Build: flutter build / gradle assembleTeacherRelease -Ptarget=lib/main_teacher.dart (student: assembleStudentRelease -Ptarget=lib/main_student.dart).
Signing: android/app/key.properties and the keystore are NOT in this repo (create your own). The Supabase key in lib/src/core.dart is the public publishable key.
Font: Cairo (OFL), place Cairo[slnt,wght].ttf at assets/fonts/Cairo.ttf (not included here).
Web app and backend schema: see the repo root.

class FadeSlideIn extends StatefulWidget {
  final Widget child;
  final int delayMs;
  const FadeSlideIn({super.key, required this.child, this.delayMs = 0});
  @override
  State<FadeSlideIn> createState() => _FadeSlideInState();

class _FadeSlideInState extends State<FadeSlideIn> with SingleTickerProviderStateMixin {
  late final AnimationController _c;
  late final Animation<double> _fade;
  late final Animation<Offset> _slide;
  @override
  void initState() {
    super.initState();
    _c = AnimationController(vsync: this, duration: const Duration(milliseconds: 600));
    _fade = CurvedAnimation(parent: _c, curve: Curves.easeOut);
    _slide = Tween<Offset>(begin: const Offset(0, 0.12), end: Offset.zero).animate(CurvedAnimation(parent: _c, curve: Curves.easeOutCubic));
    Future.delayed(Duration(milliseconds: widget.delayMs), () { if (mounted) _c.forward(); });
  }
  @override
  void dispose() { _c.dispose(); super.dispose(); }
  @override
  Widget build(BuildContext context) => FadeTransition(opacity: _fade, child: SlideTransition(position: _slide, child: widget.child));
}

// ---------- In-app update checker ----------
// Fetches app-version.json from the site; if a newer versionCode exists, offers
// a one-tap update (downloads the APK; installs over the current app - no uninstall).
const kVersionUrl = 'https://adhamahmedr99-droid.github.io/exam-platform/app-version.json';
Future<void> checkForUpdate(BuildContext context, String appKey) async {
  try {
    final info = await PackageInfo.fromPlatform();
    final res = await http.get(Uri.parse(kVersionUrl)).timeout(const Duration(seconds: 8));
    if (res.statusCode != 200) return;
    final j = jsonDecode(res.body);
    if (j is! Map || j[appKey] is! Map) return;
    final m = j[appKey] as Map;
    final latest = (m['versionCode'] as num?)?.toInt() ?? 0;
    final current = int.tryParse(info.buildNumber) ?? 0;
    if (latest <= current || !context.mounted) return;
    final url = m['apk']?.toString() ?? '';
    showDialog(
      context: context,
      barrierDismissible: false,
      builder: (ctx) => Directionality(textDirection: TextDirection.rtl, child: _UpdateDialog(url: url, versionName: '${m['versionName'] ?? ''}')),
    );
  } catch (_) {/* offline or no update info: stay quiet */}
}

class _UpdateDialog extends StatefulWidget {
  final String url, versionName;
  const _UpdateDialog({required this.url, required this.versionName});
  @override
  State<_UpdateDialog> createState() => _UpdateDialogState();
}

class _UpdateDialogState extends State<_UpdateDialog> {
  bool _started = false;
  int _pct = 0;
  String _msg = '';
  bool _failed = false;

  void _go() {
    setState(() { _started = true; _failed = false; _msg = 'جاري تنزيل التحديث…'; _pct = 0; });
    try {
      OtaUpdate().execute(widget.url, destinationFilename: 'manassa-update.apk').listen((e) {
        if (!mounted) return;
        switch (e.status) {
          case OtaStatus.DOWNLOADING:
            setState(() { _pct = int.tryParse(e.value ?? '') ?? _pct; _msg = 'جاري تنزيل التحديث… $_pct%'; });
            break;
          case OtaStatus.INSTALLING:
            setState(() => _msg = 'اضغط "تثبيت" في الشاشة اللي ظهرت. لو اتسألت، اسمح بالتثبيت من التطبيق ده.');
            break;
          case OtaStatus.INSTALLATION_DONE:
            Navigator.of(context).maybePop();
            break;
          case OtaStatus.PERMISSION_NOT_GRANTED_ERROR:
            setState(() { _failed = true; _msg = 'محتاج تسمح للتطبيق بتثبيت التحديثات من الإعدادات، وبعدها جرّب تاني.'; });
            break;
          default:
            setState(() { _failed = true; _msg = 'التحديث ماتنزلش. اتأكد من النت وجرّب تاني، أو حمّله من اللينك.'; });
        }
      }, onError: (_) { if (mounted) setState(() { _failed = true; _msg = 'التحديث ماتنزلش. اتأكد من النت وجرّب تاني، أو حمّله من اللينك.'; }); });
    } catch (_) {
      setState(() { _failed = true; _msg = 'التحديث ماتنزلش. جرّب تاني.'; });
    }
  }

  @override
  Widget build(BuildContext context) {
    return AlertDialog(
      title: const Text('في تحديث جديد 🎉'),
      content: Column(mainAxisSize: MainAxisSize.min, crossAxisAlignment: CrossAxisAlignment.start, children: [
        Text(_started ? _msg : 'النسخة ${widget.versionName} متاحة. التحديث بينزل جوه التطبيق وبيتثبت فوق الحالي، من غير ما تمسح التطبيق ولا تضيع أي بيانات.'),
        if (_started && !_failed) ...[
          const SizedBox(height: 14),
          LinearProgressIndicator(value: _pct > 0 ? _pct / 100 : null, minHeight: 8, color: kTeal, backgroundColor: const Color(0xFFEADFC9)),
        ],
      ]),
      actions: [
        if (!_started || _failed) TextButton(onPressed: () => Navigator.pop(context), child: const Text('بعدين')),
        if (_failed) TextButton(onPressed: () => launchUrl(Uri.parse(widget.url), mode: LaunchMode.externalApplication), child: const Text('حمّل من اللينك')),
        if (!_started || _failed) ElevatedButton(onPressed: _go, style: ElevatedButton.styleFrom(minimumSize: const Size(110, 46)), child: Text(_failed ? 'جرّب تاني' : 'حدّث دلوقتي')),
      ],
    );
  }
}
