import 'package:flutter/material.dart';
import 'src/core.dart';
import 'teacher/auth.dart';
import 'teacher/home.dart';

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await initSupabase();
  runApp(const TeacherApp());
}

class TeacherApp extends StatefulWidget {
  const TeacherApp({super.key});
  @override
  State<TeacherApp> createState() => _TeacherAppState();
}

class _TeacherAppState extends State<TeacherApp> {
  bool _loggedIn = false;

  @override
  void initState() {
    super.initState();
    _loggedIn = sb.auth.currentUser != null;
    WidgetsBinding.instance.addPostFrameCallback((_) => checkForUpdate('teacher'));
    sb.auth.onAuthStateChange.listen((e) {
      final logged = e.session != null;
      if (mounted && logged != _loggedIn) setState(() => _loggedIn = logged);
    });
  }

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      title: 'المنصة',
      debugShowCheckedModeBanner: false,
      navigatorKey: navKey,
      builder: (context, child) => Directionality(textDirection: TextDirection.rtl, child: child ?? const SizedBox()),
      theme: buildTheme(teacher: true),
      home: _loggedIn ? const TeacherHome() : TeacherLoginPage(onLoggedIn: () => setState(() => _loggedIn = true)),
    );
  }
}

import 'package:flutter/material.dart';
import '../src/core.dart';

class AnalyticsPage extends StatefulWidget {
  const AnalyticsPage({super.key});
  @override
  State<AnalyticsPage> createState() => _AnalyticsPageState();
}

class _AnalyticsPageState extends State<AnalyticsPage> {
  List _exams = [];
  List _subs = [];
  List _questions = [];
  Map<String, num> _gsum = {};
  Map? _exam;
  bool _loading = true;

  @override
  void initState() { super.initState(); _load(); }

  Future<void> _load() async {
    setState(() => _loading = true);
    try {
      final exams = await sb.from('exams').select('id,title,pass_pct').order('created_at', ascending: false);
      final subs = await sb.from('submissions')
          .select('*, exam_sessions(code_id, codes(student_name, exam_id))')
          .order('submitted_at', ascending: false);
      final grades = await sb.from('essay_grades').select('*');
      final gsum = <String, num>{};
      for (final g in grades) { gsum[g['submission_id']] = (gsum[g['submission_id']] ?? 0) + (g['awarded'] as num); }
      if (!mounted) return;
      _exams = exams; _subs = subs; _gsum = gsum;
      final withRes = exams.where((e) => subs.any((s) => s['exam_sessions']?['codes']?['exam_id'] == e['id'])).toList();
      if (withRes.isEmpty) { setState(() { _exam = null; _loading = false; }); return; }
      final exam = withRes.firstWhere((e) => _exam != null && e['id'] == _exam!['id'], orElse: () => withRes.first);
      final qs = await sb.from('questions').select('id,text,type,points').eq('exam_id', exam['id']).order('ord');
      if (mounted) setState(() { _exam = exam; _questions = qs; _loading = false; });
    } catch (_) { if (mounted) { setState(() => _loading = false); toast(context, 'حصلت مشكلة في التحميل', err: true); } }
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(title: const Text('التحليلات')),
        body: _loading
            ? const Center(child: CircularProgressIndicator())
            : _exam == null
                ? Center(child: Text('مفيش بيانات لسه، أول ما تيجي نتايج هتظهر هنا.', textAlign: TextAlign.center, style: TextStyle(color: Colors.grey.shade600)))
                : _body(),
      ),
    );
  }

  Widget _body() {
    final examId = _exam!['id'];
    final rows = _subs.where((s) => s['exam_sessions']?['codes']?['exam_id'] == examId).toList();
    final essayMax = _questions.where((q) => q['type'] == 'essay').fold<num>(0, (a, q) => a + (q['points'] as num));
    num finalPct(Map s) {
      final t = (s['auto_total'] as num) + essayMax;
      return t == 0 ? 0 : (((s['auto_score'] as num) + (_gsum[s['id']] ?? 0)) * 1000 / t).round() / 10;
    }
    final pcts = rows.map((s) => finalPct(s as Map)).toList();
    final avg = pcts.isEmpty ? 0 : (pcts.fold<num>(0, (a, b) => a + b) / pcts.length).round();
    final passRate = pcts.isEmpty ? 0 : (pcts.where((p) => p >= (_exam!['pass_pct'] as num)).length * 100 / pcts.length).round();
    final auto = _questions.where((q) => q['type'] != 'essay').toList();
    final qStats = auto.asMap().entries.map((e) {
      final i = e.key;
      final att = rows.where((r) => (r['bitmap'] ?? '').length > i).toList();
      final ok = att.where((r) => r['bitmap'][i] == '1').length;
      return {'q': e.value, 'pct': att.isEmpty ? 0 : (ok * 100 / att.length).round(), 'n': att.length};
    }).toList();
    final ranked = rows.toList()..sort((a, b) => finalPct(b).compareTo(finalPct(a)));
    final missed = qStats.toList()..sort((a, b) => (a['pct'] as int).compareTo(b['pct'] as int));
    final withRes = _exams.where((e) => _subs.any((s) => s['exam_sessions']?['codes']?['exam_id'] == e['id'])).toList();
    return ListView(padding: const EdgeInsets.all(16), children: [
      if (withRes.length > 1)
        DropdownButtonFormField<String>(
          value: _exam!['id'] as String,
          decoration: const InputDecoration(labelText: 'الامتحان'),
          items: withRes.map<DropdownMenuItem<String>>((e) => DropdownMenuItem(value: e['id'] as String, child: Text(e['title'] ?? ''))).toList(),
          onChanged: (v) { setState(() => _exam = _exams.firstWhere((e) => e['id'] == v)); _load(); },
        ),
      const SizedBox(height: 8),
      Row(children: [_stat('تسليم', '${rows.length}'), _stat('المتوسط', '$avg%'), _stat('أعلى', '${pcts.isEmpty ? 0 : pcts.reduce((a, b) => a > b ? a : b)}%')]),
      Row(children: [_stat('أقل', '${pcts.isEmpty ? 0 : pcts.reduce((a, b) => a < b ? a : b)}%'), _stat('نسبة النجاح', '$passRate%'), _stat('أسئلة', '${_questions.length}')]),
      const SizedBox(height: 8),
      Card(child: Padding(
        padding: const EdgeInsets.all(14),
        child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
          const Text('ترتيب الطلبة', style: TextStyle(fontWeight: FontWeight.w800, fontSize: 16)),
          const SizedBox(height: 8),
          ...ranked.asMap().entries.map((e) {
            final s = e.value;
            final p = finalPct(s);
            return Padding(
              padding: const EdgeInsets.symmetric(vertical: 5),
              child: Row(children: [
                SizedBox(width: 26, child: Text('${e.key + 1}.', style: TextStyle(color: Colors.grey.shade500))),
                Expanded(child: Text(s['exam_sessions']?['codes']?['student_name'] ?? '—', style: const TextStyle(fontWeight: FontWeight.w700))),
                Text('$p%', style: const TextStyle(fontWeight: FontWeight.w900)),
              ]),
            );
          }),
        ]),
      )),
      Card(child: Padding(
        padding: const EdgeInsets.all(14),
        child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
          const Text('صعوبة الأسئلة (نسبة الصح)', style: TextStyle(fontWeight: FontWeight.w800, fontSize: 16)),
          const SizedBox(height: 10),
          ...qStats.asMap().entries.map((e) {
            final st = e.value;
            final qt = (st['q']['text'] ?? '').toString();
            final pctv = st['pct'] as int;
            return Padding(
              padding: const EdgeInsets.symmetric(vertical: 6),
              child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
                Text('س${e.key + 1}: ${qt.length > 50 ? '${qt.substring(0, 49)}…' : qt}', style: const TextStyle(fontSize: 13, fontWeight: FontWeight.w600)),
                const SizedBox(height: 4),
                Row(children: [
                  Expanded(child: ClipRRect(
                    borderRadius: BorderRadius.circular(6),
                    child: LinearProgressIndicator(value: pctv / 100, minHeight: 8, backgroundColor: Colors.grey.shade200, color: pctv >= 60 ? kTeal : (pctv >= 40 ? kGold : Colors.red)),
                  )),
                  const SizedBox(width: 8),
                  Text('$pctv% • ${st['n']}', style: TextStyle(fontSize: 11.5, color: Colors.grey.shade600)),
                ]),
              ]),
            );
          }),
        ]),
      )),
      Card(child: Padding(
        padding: const EdgeInsets.all(14),
        child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
          const Text('أكتر أسئلة الطلبة غلطوا فيها', style: TextStyle(fontWeight: FontWeight.w800, fontSize: 16)),
          const SizedBox(height: 8),
          if (missed.isEmpty) Text('مفيش بيانات كفاية.', style: TextStyle(color: Colors.grey.shade600)),
          ...missed.take(3).map((m) {
            final qt = (m['q']['text'] ?? '').toString();
            return ListTile(
              dense: true, contentPadding: EdgeInsets.zero,
              leading: Icon(Icons.error_outline, color: Colors.red.shade300),
              title: Text(qt.length > 70 ? '${qt.substring(0, 69)}…' : qt, style: const TextStyle(fontSize: 13)),
              trailing: Text('${m['pct']}% صح', style: const TextStyle(fontWeight: FontWeight.w700, fontSize: 12.5)),
            );
          }),
        ]),
      )),
    ]);
  }

  Widget _stat(String label, String v) => Expanded(
        child: Card(
          margin: const EdgeInsets.all(3),
          child: Padding(
            padding: const EdgeInsets.symmetric(vertical: 14),
            child: Column(children: [
              Text(v, style: const TextStyle(fontSize: 19, fontWeight: FontWeight.w900, color: kNavy)),
              Text(label, style: TextStyle(fontSize: 11, color: Colors.grey.shade600)),
            ]),
          ),
        ),
      );
}

import 'package:flutter/material.dart';
import 'package:supabase_flutter/supabase_flutter.dart';
import '../src/core.dart';

class TeacherLoginPage extends StatefulWidget {
  final VoidCallback onLoggedIn;
  const TeacherLoginPage({super.key, required this.onLoggedIn});
  @override
  State<TeacherLoginPage> createState() => _TeacherLoginPageState();
}

class _TeacherLoginPageState extends State<TeacherLoginPage> {
  final _email = TextEditingController();
  final _pass = TextEditingController();
  bool _busy = false;
  bool _signupMode = false;

  Future<void> _submit() async {
    final email = _email.text.trim(), pass = _pass.text;
    if (email.isEmpty || pass.isEmpty) { toast(context, 'اكتب الإيميل وكلمة السر', err: true); return; }
    setState(() => _busy = true);
    try {
      if (_signupMode) {
        await sb.auth.signUp(email: email, password: pass);
        if (mounted) toast(context, 'الحساب اتعمل. سجّل دخول دلوقتي بنفس الإيميل وكلمة السر.');
        setState(() => _signupMode = false);
      } else {
        await sb.auth.signInWithPassword(email: email, password: pass);
        widget.onLoggedIn();
      }
    } on AuthException catch (e) {
      if (mounted) toast(context, (e.statusCode == '429' || e.message.toLowerCase().contains('rate limit')) ? 'محاولات كتير. استنى شوية وجرّب تاني.' : (_signupMode ? 'مشكلة في إنشاء الحساب: ${e.message}' : 'الإيميل أو كلمة السر مش مظبوطين'), err: true);
    } catch (_) {
      if (mounted) toast(context, 'حصلت مشكلة، جرب تاني', err: true);
    }
    if (mounted) setState(() => _busy = false);
  }

  Future<void> _forgot() async {
    final email = _email.text.trim();
    if (email.isEmpty) { toast(context, 'اكتب إيميلك الأول وبعدين دوس نسيت كلمة السر', err: true); return; }
    setState(() => _busy = true);
    try {
      await sb.auth.resetPasswordForEmail(email);
      if (mounted) toast(context, 'بعنالك لينك إعادة تعيين كلمة السر على إيميلك');
    } catch (_) {
      if (mounted) toast(context, 'حصلت مشكلة، جرب تاني', err: true);
    }
    if (mounted) setState(() => _busy = false);
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        body: SingleChildScrollView(
          child: Column(children: [
            FadeSlideIn(child: const BrandHeader(subtitle: 'امتحانات وفيديوهات شرح طلبتك في مكان واحد')),
            Padding(
              padding: const EdgeInsets.all(20),
              child: FadeSlideIn(
                delayMs: 200,
                child: Card(
                  child: Padding(
                    padding: const EdgeInsets.all(20),
                    child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
                      Text(_signupMode ? 'حساب مدرس جديد' : 'دخول المدرس',
                          style: const TextStyle(fontSize: 20, fontWeight: FontWeight.w800), textAlign: TextAlign.center),
                      const SizedBox(height: 6),
                      Text('كل مدرس ليه حسابه الخاص: امتحاناته وطلبته ودرجاتهم منفصلة تمامًا.',
                          style: TextStyle(color: Colors.grey.shade600, fontSize: 13), textAlign: TextAlign.center),
                      const SizedBox(height: 20),
                      TextField(
                        controller: _email,
                        keyboardType: TextInputType.emailAddress,
                        textDirection: TextDirection.ltr,
                        textAlign: TextAlign.left,
                        decoration: const InputDecoration(labelText: 'البريد الإلكتروني', prefixIcon: Icon(Icons.mail_outline)),
                      ),
                      const SizedBox(height: 12),
                      PwField(controller: _pass, action: TextInputAction.done, onSubmit: (_) => _submit()),
                      Align(
                        alignment: AlignmentDirectional.centerStart,
                        child: TextButton(onPressed: _busy ? null : _forgot, child: const Text('نسيت كلمة السر؟')),
                      ),
                      BusyBtn(label: _signupMode ? 'إنشاء الحساب' : 'دخول', busy: _busy, onTap: _submit, icon: Icons.login),
                      TextButton(
                        onPressed: _busy ? null : () => setState(() => _signupMode = !_signupMode),
                        child: Text(_signupMode ? 'عندك حساب؟ سجّل دخول' : 'أول مرة؟ اعمل حساب جديد'),
                      ),
                      Text('حساب جديد بيدخل على طول من غير تأكيد إيميل.',
                          style: TextStyle(color: Colors.grey.shade500, fontSize: 12), textAlign: TextAlign.center),
                    ]),
                  ),
                ),
              ),
            ),
          ]),
        ),
      ),
    );
  }
}

import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:url_launcher/url_launcher.dart';
import '../src/core.dart';

class CodesPage extends StatefulWidget {
  const CodesPage({super.key});
  @override
  State<CodesPage> createState() => _CodesPageState();
}

class _CodesPageState extends State<CodesPage> {
  List _exams = [];
  Map? _exam;
  List _codes = [];
  List _students = [];
  bool _loading = true;
  bool _busy = false;

  @override
  void initState() { super.initState(); _loadExams(); }

  Future<void> _loadExams() async {
    setState(() => _loading = true);
    try {
      final e = await sb.from('exams').select('id,title,published,open_enabled,open_token').order('created_at', ascending: false);
      final st = await sb.from('students').select('*').order('name');
      if (!mounted) return;
      setState(() { _exams = e; _students = st; _exam = e.isEmpty ? null : e.first; });
      await _loadCodes();
    } catch (_) { if (mounted) { setState(() => _loading = false); toast(context, 'حصلت مشكلة في التحميل', err: true); } }
  }

  Future<void> _loadCodes() async {
    if (_exam == null) { if (mounted) setState(() => _loading = false); return; }
    try {
      final c = await sb.rpc('exam_code_status', params: {'p_exam': _exam!['id']});
      if (mounted) setState(() { _codes = c; _loading = false; });
    } catch (_) { if (mounted) setState(() => _loading = false); }
  }

  String _chipFor(String? s) => s == 'completed' ? 'مكتمل' : s == 'in_progress' ? 'قيد الحل دلوقتي' : 'غير مستخدم';
  Color _chipColor(String? s) => s == 'completed' ? kTeal : s == 'in_progress' ? kGold : Colors.grey;

  String _invite(Map c) =>
      'أهلًا ${c['student_name']}، ده الكود بتاعك لامتحان "${_exam!['title']}". افتح اللينك ده واكتب الكود عشان تبدأ: $kSiteUrl\n\nالكود: ${c['token']}';

  Future<void> _publish() async {
    setState(() => _busy = true);
    try {
      await sb.from('exams').update({'published': true}).eq('id', _exam!['id']);
      await _loadExams();
      if (mounted) { setState(() { _exam = _exams.firstWhere((e) => e['id'] == _exam!['id'], orElse: () => _exam!); }); toast(context, 'الامتحان اتنشر'); }
    } catch (_) { if (mounted) toast(context, 'النشر متمش، جرب تاني', err: true); }
    if (mounted) setState(() => _busy = false);
  }

  Future<void> _genFromNames() async {
    final ctrl = TextEditingController();
    final ok = await showDialog<bool>(
      context: context,
      builder: (ctx) => Directionality(
        textDirection: TextDirection.rtl,
        child: AlertDialog(
          title: const Text('أسماء الطلبة'),
          content: TextField(controller: ctrl, maxLines: 6, decoration: const InputDecoration(hintText: 'كل اسم في سطر', border: OutlineInputBorder())),
          actions: [
            TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('إلغاء')),
            ElevatedButton(onPressed: () => Navigator.pop(ctx, true), child: const Text('اعمل الأكواد')),
          ],
        ),
      ),
    );
    if (ok != true) return;
    final names = ctrl.text.split('\n').map((s) => s.trim()).where((s) => s.isNotEmpty && !_codes.any((c) => c['student_name'] == s)).toList();
    await _gen(names);
  }

  Future<void> _genForRegistered() async {
    final names = _students.map((s) => s['name'].toString()).where((n) => !_codes.any((c) => c['student_name'] == n)).toList();
    await _gen(names);
  }

  Future<void> _gen(List<String> names) async {
    if (names.isEmpty) { toast(context, 'مفيش أسماء جديدة تتعملها أكواد', err: true); return; }
    setState(() => _busy = true);
    try {
      final rows = names.map((n) => {
            'exam_id': _exam!['id'], 'student_name': n, 'token': newToken(),
            'student_id': _students.firstWhere((s) => s['name'] == n, orElse: () => {'id': null})['id'],
          }).toList();
      await sb.from('codes').insert(rows);
      await _loadCodes();
      if (mounted) toast(context, 'اتعمل ${names.length} كود');
    } catch (_) { if (mounted) toast(context, 'حصلت مشكلة', err: true); }
    if (mounted) setState(() => _busy = false);
  }

  Future<void> _openLink(String action) async {
    try {
      if (action == 'enable') {
        await sb.from('exams').update({'open_enabled': true, 'open_token': _exam!['open_token'] ?? newToken()}).eq('id', _exam!['id']);
      } else if (action == 'disable') {
        await sb.from('exams').update({'open_enabled': false}).eq('id', _exam!['id']);
      } else {
        await sb.from('exams').update({'open_token': newToken()}).eq('id', _exam!['id']);
      }
      await _loadExams();
      if (mounted) setState(() { _exam = _exams.firstWhere((e) => e['id'] == _exam!['id'], orElse: () => _exam!); });
    } catch (_) { if (mounted) toast(context, 'حصلت مشكلة', err: true); }
  }

  @override
  Widget build(BuildContext context) {
    final openEnabled = _exam?['open_enabled'] == true;
    final openLink = '$kSiteUrl?open=${_exam?['open_token'] ?? ''}';
    final waText = 'لينك امتحان "${_exam?['title'] ?? ''}": افتح اللينك، اكتب اسمك، وابدأ على طول: $openLink';
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(title: const Text('أكواد الطلبة')),
        body: _loading
            ? const Center(child: CircularProgressIndicator())
            : _exam == null
                ? Center(child: Text('اعمل امتحان الأول من صفحة الامتحانات.', style: TextStyle(color: Colors.grey.shade600)))
                : RefreshIndicator(
                    onRefresh: _loadExams,
                    child: ListView(padding: const EdgeInsets.all(16), children: [
                      DropdownButtonFormField<String>(
                        value: _exam!['id'] as String,
                        decoration: const InputDecoration(labelText: 'الامتحان'),
                        items: _exams.map<DropdownMenuItem<String>>((e) => DropdownMenuItem(value: e['id'] as String, child: Text(e['title'] ?? ''))).toList(),
                        onChanged: (v) { setState(() { _exam = _exams.firstWhere((e) => e['id'] == v); _loading = true; }); _loadCodes(); },
                      ),
                      const SizedBox(height: 12),
                      if (_exam!['published'] != true) ...[
                        Card(child: Padding(
                          padding: const EdgeInsets.all(16),
                          child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
                            const Text('الامتحان ده محفوظ كمسودة. انشره الأول عشان تقدر تعمل أكواد.'),
                            const SizedBox(height: 10),
                            BusyBtn(label: '١. انشر الامتحان', busy: _busy, onTap: _publish, icon: Icons.publish),
                          ]),
                        )),
                      ] else ...[
                        Card(child: Padding(
                          padding: const EdgeInsets.all(16),
                          child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
                            const Text('١. اعمل الأكواد', style: TextStyle(fontWeight: FontWeight.w800, fontSize: 16)),
                            const SizedBox(height: 4),
                            Text('كل طالب ليه كود مختلف. ابعت لكل طالب دعوته بس.', style: TextStyle(color: Colors.grey.shade600, fontSize: 13)),
                            const SizedBox(height: 12),
                            ElevatedButton.icon(onPressed: _busy ? null : _genFromNames, icon: const Icon(Icons.playlist_add), label: const Text('اكتب أسماء وولّد')),
                            const SizedBox(height: 8),
                            OutlinedButton.icon(onPressed: _busy ? null : _genForRegistered, icon: const Icon(Icons.groups_outlined), label: const Text('ولّد للطلبة المسجلين')),
                          ]),
                        )),
                        Card(child: Padding(
                          padding: const EdgeInsets.all(16),
                          child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
                            const Text('٢. أو شارك لينك مفتوح (من غير كود)', style: TextStyle(fontWeight: FontWeight.w800, fontSize: 16)),
                            const SizedBox(height: 8),
                            if (openEnabled) ...[
                              SelectableText(openLink, textDirection: TextDirection.ltr, style: const TextStyle(fontSize: 12.5)),

  bool _importing = false;

  Future<void> _importFromJson() async {
    final picked = await FilePicker.platform.pickFiles(type: FileType.any, withData: true);
    if (picked == null || picked.files.isEmpty) return;
    final f = picked.files.first;
    setState(() => _importing = true);
    try {
      var bytes = f.bytes;
      if (bytes == null && f.path != null) bytes = await File(f.path!).readAsBytes();
      if (bytes == null) throw 'مقدرتش أقرا الملف.';
      if (bytes.length > 500000) throw 'الملف كبير جدًا. الحد الأقصى ٥٠٠ كيلوبايت.';
      dynamic data;
      try { data = jsonDecode(utf8.decode(bytes)); } catch (_) { throw 'الملف مش JSON صالح. اختار ملف الامتحان الأصلي.'; }
      final rows = data is List ? data : (data is Map ? data['questions'] : null);
      if (rows is! List || rows.isEmpty || rows.length > 100) throw 'لازم الملف يحتوي من ١ إلى ١٠٠ سؤال.';
      final out = <Map<String, dynamic>>[];
      for (var i = 0; i < rows.length; i++) {
        final q = rows[i];
        if (q is! Map || q['text'] is! String || (q['text'] as String).trim().isEmpty) throw 'نص السؤال ${i + 1} ناقص.';
        if (q['type'] != null && q['type'] != 'mcq') throw 'الاستيراد بيدعم الاختيار من متعدد بس.';
        final ch = q['choices'];
        if (ch is! List || ch.length < 2 || ch.length > 6 || ch.any((c) => c is! String || c.trim().isEmpty)) throw 'راجع اختيارات السؤال ${i + 1}.';
        final c = q['correct'];
        if (c is! int || c < 0 || c >= ch.length) throw 'الإجابة الصحيحة للسؤال ${i + 1} مش محددة صح.';
        final pts = q['points'] ?? 1;
        if (pts is! num || pts < 0.5 || pts > 100) throw 'درجة السؤال ${i + 1} غير مدعومة.';
        out.add({'text': (q['text'] as String).trim(), 'choices': ch.map((e) => (e as String).trim()).toList(), 'correct': c, 'points': pts});
      }
      var ord = _questions.length;
      for (final q in out) {
        await sb.from('questions').insert({
          'exam_id': _exam['id'], 'ord': ord, 'text': q['text'], 'type': 'mcq',
          'points': q['points'], 'choices': q['choices'], 'correct': q['correct'], 'accepted': null,
        });
        ord++;
      }
      if (mounted) toast(context, 'اتضاف ${out.length} سؤال من الملف. راجعهم.');
      _load();
    } catch (e) {
      if (mounted) toast(context, e is String ? e : 'حصلت مشكلة في الاستيراد: $e', err: true);
    } finally {
      if (mounted) setState(() => _importing = false);
    }
  }

  Future<void> _importFromPhotos() async {
    final files = await ImagePicker().pickMultiImage(maxWidth: 1600, imageQuality: 70);
    if (files.isEmpty) return;
    setState(() => _importing = true);
    try {
      final picked = files.take(6).toList();
      final rows = <dynamic>[];
      final failed = <int>[];
      String lastErr = '';
      var done = 0;
      Future<void> one(int idx) async {
        try {
          final b64 = base64Encode(await picked[idx].readAsBytes());
          final res = await sb.functions.invoke('exam-from-photo', body: {'images': [b64]}).timeout(const Duration(seconds: 95));
          final out = res.data;
          if (out is Map && out['questions'] is List) rows.addAll(out['questions'] as List);
        } catch (e) {
          failed.add(idx + 1);
          final t = '$e';
          lastErr = t.contains('Timeout') ? 'الصورة خدت وقت طويل، جرّب صورة أوضح أو أصغر.'
              : t.contains('429') ? 'الخدمة مزدحمة، استنى دقيقة وجرّب تاني.'
              : (t.contains('SocketException') || t.contains('ClientException')) ? 'مفيش اتصال بالنت أو النت ضعيف.'
              : t.contains('401') ? 'سجّل دخول تاني وجرّب.' : 'الخدمة مش متاحة دلوقتي، جرّب تاني بعد شوية.';
        }
        done++;
        if (mounted) { ScaffoldMessenger.of(context).hideCurrentSnackBar(); toast(context, 'خلّصت $done من ${picked.length} صورة'); }
      }
      var next = 0;
      Future<void> worker() async { while (next < picked.length) { final i = next++; await one(i); } }
      await Future.wait([worker(), worker()]);
      if (rows.isEmpty && failed.isNotEmpty) throw lastErr;
      if (rows.isEmpty) throw 'مش لاقي أسئلة اختيار من متعدد في الصور. صوّر أوضح وجرّب تاني.';
      var ord = _questions.length;
      var added = 0;
      final guessed = <int>[];
      for (final q in rows) {
        if (q is! Map || q['choices'] is! List) continue;
        final choices = (q['choices'] as List).map((c) => '$c').toList();
        if (choices.length < 2) continue;
        var correct = q['correct'] is int ? q['correct'] as int : (int.tryParse('${q['correct']}') ?? 0);
        if (correct < 0 || correct >= choices.length) correct = 0;
        if (q['answer_source'] == 'guessed') guessed.add(ord + 1);
        await sb.from('questions').insert({
          'exam_id': _exam['id'], 'ord': ord, 'text': '${q['text'] ?? ''}', 'type': 'mcq',
          'points': 1, 'choices': choices, 'correct': correct, 'accepted': null,
        });
        ord++; added++;
      }
      if (added == 0) throw 'مش لاقي أسئلة صالحة في الصور. جرّب تاني.';
      if (mounted) {
        await showDialog(context: context, builder: (_) => Directionality(
          textDirection: TextDirection.rtl,
          child: AlertDialog(
            title: const Text('اتضافت الأسئلة'),
            content: Text('اتضاف $added سؤال من الصور.' + (failed.isEmpty ? '' : '\n\nالصور رقم ${failed.join('، ')} ماتقراتش ($lastErr) جرّبهم تاني.') + (guessed.isEmpty ? '' : '\n\nالإجابة الصحيحة للأسئلة رقم ${guessed.join('، ')} مكانتش معلّمة في الصورة والذكاء الاصطناعي اختارها — راجعها.') + '\n\nراجع الأسئلة والاختيارات كويس (القراءة من الصور ممكن تغلط).'),
            actions: [TextButton(onPressed: () => Navigator.pop(context), child: const Text('تمام'))],
          ),
        ));
      }
      _load();
    } catch (e) {

class QuestionSheet extends StatefulWidget {
  final String examId;
  final Map? question;
  final int ord;
  const QuestionSheet({super.key, required this.examId, this.question, required this.ord});
  @override
  State<QuestionSheet> createState() => _QuestionSheetState();
}

class _QuestionSheetState extends State<QuestionSheet> {
  final _text = TextEditingController();
  final _points = TextEditingController(text: '1');
  final _accepted = TextEditingController();
  String _type = 'mcq';
  List<TextEditingController> _choices = [TextEditingController(), TextEditingController()];
  int _correct = 0;
  bool _busy = false;

  @override
  void initState() {
    super.initState();
    final q = widget.question;
    if (q != null) {
      _text.text = q['text'] ?? ''; _type = q['type'] ?? 'mcq';
      _points.text = '${q['points'] ?? 1}';
      if (_type == 'mcq' && q['choices'] is List) {
        _choices = (q['choices'] as List).map((c) => TextEditingController(text: '$c')).toList();
        if (_choices.length < 2) while (_choices.length < 2) { _choices.add(TextEditingController()); }
        _correct = q['correct'] ?? 0;
      }
      if (_type == 'sa' && q['accepted'] is List) _accepted.text = (q['accepted'] as List).join('، ');
      if (_type == 'tf') _correct = q['correct'] ?? 0;
    }
  }

  Future<void> _save() async {
    if (_text.text.trim().isEmpty) { toast(context, 'اكتب نص السؤال', err: true); return; }
    if (_type == 'mcq' && _choices.where((c) => c.text.trim().isNotEmpty).length < 2) {
      toast(context, 'لازم اختيارين على الأقل', err: true); return;
    }
    setState(() => _busy = true);
    final row = <String, dynamic>{
      'exam_id': widget.examId, 'text': _text.text.trim(), 'type': _type,
      'points': (num.tryParse(_points.text) ?? 1).clamp(0.5, 100),
      'choices': null, 'correct': null, 'accepted': null,
    };
    if (_type == 'mcq') {
      final cs = _choices.map((c) => c.text.trim()).where((c) => c.isNotEmpty).toList();
      row['choices'] = cs;
      row['correct'] = _correct.clamp(0, cs.length - 1);
    } else if (_type == 'tf') {
      row['choices'] = ['صح', 'غلط'];
      row['correct'] = _correct;
    } else if (_type == 'sa') {
      row['accepted'] = _accepted.text.split(RegExp('[،,]')).map((s) => s.trim()).where((s) => s.isNotEmpty).toList();
    }
    try {
      if (widget.question == null) {
        row['ord'] = widget.ord;
        await sb.from('questions').insert(row);
      } else {
        await sb.from('questions').update(row).eq('id', widget.question!['id']);
      }
      if (mounted) Navigator.pop(context, true);
    } catch (_) {
      if (mounted) { toast(context, 'حصلت مشكلة في الحفظ', err: true); setState(() => _busy = false); }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Padding(
        padding: const EdgeInsets.all(20),
        child: ListView(shrinkWrap: true, children: [
          Text(widget.question == null ? 'سؤال جديد' : 'تعديل السؤال', style: const TextStyle(fontSize: 18, fontWeight: FontWeight.w800)),
          const SizedBox(height: 12),
          SegmentedButton<String>(
            segments: const [
              ButtonSegment(value: 'mcq', label: Text('اختياري')),
              ButtonSegment(value: 'tf', label: Text('صح/غلط')),
              ButtonSegment(value: 'sa', label: Text('قصيرة')),
              ButtonSegment(value: 'essay', label: Text('مقالي')),
            ],
            selected: {_type},
            onSelectionChanged: (s) => setState(() => _type = s.first),
          ),
          const SizedBox(height: 12),
          TextField(controller: _text, maxLines: 3, decoration: const InputDecoration(labelText: 'نص السؤال *')),
          const SizedBox(height: 12),
          if (_type == 'mcq') ...[
            const Text('الاختيارات (علّم على الإجابة الصح):', style: TextStyle(fontWeight: FontWeight.w700)),
            ...List.generate(_choices.length, (i) => Row(children: [
                  Radio<int>(value: i, groupValue: _correct, onChanged: (v) => setState(() => _correct = v!)),
                  Expanded(child: TextField(controller: _choices[i], decoration: InputDecoration(labelText: 'اختيار ${i + 1}'))),
                  if (_choices.length > 2)
                    IconButton(icon: const Icon(Icons.remove_circle_outline, color: Colors.red), onPressed: () => setState(() => _choices.removeAt(i))),
                ])),
            TextButton.icon(onPressed: () => setState(() => _choices.add(TextEditingController())), icon: const Icon(Icons.add), label: const Text('ضيف اختيار')),
          ],
          if (_type == 'tf')
            Row(children: [
              Expanded(child: RadioListTile<int>(value: 0, groupValue: _correct, onChanged: (v) => setState(() => _correct = v!), title: const Text('صح'))),
              Expanded(child: RadioListTile<int>(value: 1, groupValue: _correct, onChanged: (v) => setState(() => _correct = v!), title: const Text('غلط'))),
            ]),
          if (_type == 'sa')
            TextField(controller: _accepted, decoration: const InputDecoration(labelText: 'الإجابات المقبولة (افصل بينهم بفاصلة)')),
          const SizedBox(height: 12),
          TextField(controller: _points, keyboardType: TextInputType.number, decoration: const InputDecoration(labelText: 'الدرجة')),
          const SizedBox(height: 16),
          BusyBtn(label: 'حفظ السؤال', busy: _busy, onTap: _save, icon: Icons.check),
        ]),
      ),
    );
  }
}

      final grades = await sb.from('essay_grades').select('*');
      final gmap = <String, Map<String, num>>{};
      for (final g in grades) {
        gmap.putIfAbsent(g['submission_id'] as String, () => {})[g['question_id'] as String] = g['awarded'] as num;
      }
      final qmap = <String, List>{};
      final examIds = <String>{};
      for (final s in subs) {
        final ex = s['exam_sessions']?['codes']?['exams'];
        if (ex != null) examIds.add(ex['id'] as String);

      for (final id in examIds) {

    } catch (_) { if (mounted) { setState(() => _loading = false); toast(context, 'حصلت مشكلة في التحميل', err: true); } }

  Map<String, dynamic> _report(Map s) {
    final codes = s['exam_sessions']?['codes'] ?? {};
    final ex = codes['exams'] ?? {};
    final qs = _examQuestions[ex['id']] ?? [];
    final essays = qs.where((q) => q['type'] == 'essay').toList();
    final essayMax = essays.fold<num>(0, (a, q) => a + (q['points'] as num));
    final gm = _gmap[s['id']] ?? {};
    final awarded = gm.values.fold<num>(0, (a, b) => a + b);
    final pending = essays.isNotEmpty && essays.any((q) => !gm.containsKey(q['id']));
    final score = (s['auto_score'] as num) + awarded;
    final total = (s['auto_total'] as num) + essayMax;
    final pct = total == 0 ? 0.0 : (score * 1000 / total).round() / 10;
    final wrong = <String>[];
    final auto = qs.where((q) => q['type'] != 'essay').toList();
    final bitmap = (s['bitmap'] ?? '') as String;
    for (var i = 0; i < auto.length && i < bitmap.length; i++) {
      if (bitmap[i] == '0') wrong.add(auto[i]['text'].toString());
    }
    final level = pct >= 85 ? 'ممتاز' : pct >= 75 ? 'جيد جدًا' : pct >= 65 ? 'جيد' : pct >= 50 ? 'مقبول' : 'ضعيف - يحتاج دعم';
    final passPct = ex['pass_pct'] ?? 50;
    final date = DateTime.tryParse(s['submitted_at'] ?? '')?.toLocal();
    final lines = <String>[
      'تقرير أداء الطالب: ${codes['student_name'] ?? ''}',

class EssayGradeSheet extends StatefulWidget {
  final String submissionId;
  final List questions;
  final Map<String, dynamic> answers;
  final Map<String, num> grades;
  const EssayGradeSheet({super.key, required this.submissionId, required this.questions, required this.answers, required this.grades});
  @override
  State<EssayGradeSheet> createState() => _EssayGradeSheetState();
}

class _EssayGradeSheetState extends State<EssayGradeSheet> {
  final _ctrls = <String, TextEditingController>{};
  bool _busy = false;

  @override
  void initState() {
    super.initState();
    for (final q in widget.questions) {
      _ctrls[q['id']] = TextEditingController(text: widget.grades[q['id']]?.toString() ?? '');
    }
  }

  Future<void> _save() async {
    setState(() => _busy = true);
    try {
      for (final q in widget.questions) {
        final t = _ctrls[q['id']]!.text.trim();
        if (t.isEmpty) continue;
        final max = (q['points'] as num).toDouble();
        final v = (double.tryParse(t) ?? 0).clamp(0, max);
        await sb.from('essay_grades').upsert({'submission_id': widget.submissionId, 'question_id': q['id'], 'awarded': v});
      }
      if (mounted) { Navigator.pop(context, true); toast(context, 'التصحيح اتحفظ'); }
    } catch (_) { if (mounted) { setState(() => _busy = false); toast(context, 'حصلت مشكلة', err: true); } }
  }

  @override
  Widget build(BuildContext context) {
    return DraggableScrollableSheet(
      expand: false, initialChildSize: 0.85, maxChildSize: 0.95,
      builder: (_, sc) => ListView(controller: sc, padding: const EdgeInsets.all(20), children: [
        const Text('تصحيح المقالي', style: TextStyle(fontSize: 18, fontWeight: FontWeight.w800), textAlign: TextAlign.center),
        const SizedBox(height: 12),
        ...widget.questions.map((q) {
          final ans = widget.answers[q['id']];
          String ansText = '(الطالب مكتبش إجابة)';
          if (ans != null) {
            ansText = ans is String ? ans : (ans['text'] ?? ans.toString());
          }
          return Card(
            color: kBg,
            child: Padding(
              padding: const EdgeInsets.all(14),
              child: Column(crossAxisAlignment: CrossAxisAlignment.stretch, children: [
                Text('${q['text']} (من ${q['points']})', style: const TextStyle(fontWeight: FontWeight.w800)),
                const SizedBox(height: 8),
                Container(
                  padding: const EdgeInsets.all(12),
                  decoration: BoxDecoration(color: Colors.white, borderRadius: BorderRadius.circular(12)),
                  child: Text(ansText, style: TextStyle(color: Colors.grey.shade800, height: 1.5)),
                ),
                const SizedBox(height: 8),
                TextField(
                  controller: _ctrls[q['id']],
                  keyboardType: const TextInputType.numberWithOptions(decimal: true),
                  textDirection: TextDirection.ltr,
                  decoration: InputDecoration(labelText: 'الدرجة (حد أقصى ${q['points']})'),
                ),
              ]),
            ),
          );
        }),
        const SizedBox(height: 10),
        BusyBtn(label: 'احفظ التصحيح', busy: _busy, onTap: _save, icon: Icons.save_outlined),
      ]),
    );
  }
}

  Future<void> _load() async {
    setState(() => _loading = true);
    try {
      final exams = await sb.from('exams').select('id,title,duration_min').order('created_at', ascending: false);
      if (!mounted) return;
      if (exams.isEmpty) { setState(() { _exams = []; _exam = null; _loading = false; }); return; }
      final exam = exams.firstWhere((e) => _exam != null && e['id'] == _exam!['id'], orElse: () => exams.first);
      final codes = await sb.from('codes')
          .select('id,student_name,status,exam_sessions(id,started_at,status,submissions(id,auto_score,auto_total,time_sec,submitted_at,essay_grades(question_id,awarded)))')
          .eq('exam_id', exam['id']).order('created_at');
      final essays = await sb.from('questions').select('id,points').eq('exam_id', exam['id']).eq('type', 'essay');
      if (!mounted) return;
      setState(() { _exams = exams; _exam = exam; _rows = codes; _essays = essays; _loadedAt = DateTime.now(); _loading = false; });
    } catch (_) { if (mounted) { setState(() => _loading = false); toast(context, 'مش قادرين نحمّل المتابعة دلوقتي، جرب تاني', err: true); } }
  }

  Map<String, dynamic> _calc(Map c) {
    final sessions = (c['exam_sessions'] as List?) ?? [];
    final session = sessions.isEmpty ? null : sessions.first;
    final subs = (session?['submissions'] as List?) ?? [];
    final sub = subs.isEmpty ? null : subs.first;
    final status = (sub != null || session?['status'] == 'submitted' || c['status'] == 'completed')
        ? 'submitted'
        : (session != null || c['status'] == 'in_progress')
            ? 'in_progress'
            : 'unused';
    int? elapsed;
    if (session?['started_at'] != null) {
      elapsed = DateTime.now().difference(DateTime.parse(session!['started_at']).toLocal()).inSeconds.clamp(0, 1 << 31);
    }
    final overdue = status == 'in_progress' && elapsed != null && elapsed >= ((_exam?['duration_min'] ?? 15) as num) * 60;
    final grades = (sub?['essay_grades'] as List?) ?? [];
    final pending = sub != null && _essays.any((q) => !grades.any((g) => g['question_id'] == q['id']));
    final awarded = _essays.fold<num>(0, (s, q) {
      final g = grades.firstWhere((g) => g['question_id'] == q['id'], orElse: () => null);
      return s + ((g?['awarded'] as num?) ?? 0);
    });
    num? score, total;
    if (sub != null) {
      score = (sub['auto_score'] as num) + awarded;
      total = (sub['auto_total'] as num) + _essays.fold<num>(0, (s, q) => s + (q['points'] as num));
    }
    return {'status': status, 'elapsed': elapsed, 'overdue': overdue, 'pending': pending, 'score': score, 'total': total, 'sub': sub, 'session': session};
  }

          content: Column(mainAxisSize: MainAxisSize.min, children: [
            TextField(controller: name, decoration: const InputDecoration(labelText: 'اسم الطالب *')),
            const SizedBox(height: 10),
            TextField(controller: user, textDirection: TextDirection.ltr, textAlign: TextAlign.left,
                decoration: const InputDecoration(labelText: 'اسم المستخدم (إنجليزي من غير مسافات) *')),
            const SizedBox(height: 10),
            PwField(controller: pass, label: 'كلمة السر *'),
          ]),
          actions: [
            TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('إلغاء')),
            ElevatedButton(onPressed: () => Navigator.pop(ctx, true), child: const Text('إضافة')),
          ],
        ),
      ),
    );
    if (ok != true) return;
    try {
      final r = await sb.rpc('create_video_student', params: {
        'p_name': name.text.trim(), 'p_username': user.text.trim(), 'p_password': pass.text,
      });
      if (r['ok'] != true) {
        final m = {
          'bad_username': 'اسم المستخدم لازم يكون 3 حروف إنجليزي أو أرقام على الأقل من غير مسافات.',
          'bad_password': 'كلمة السر لازم تكون 8 حروف على الأقل.',
          'taken': 'اسم المستخدم ده متاخد، جرب اسم تاني.',
          'auth': 'سجّل دخول من الأول.',
        };
        if (mounted) toast(context, m[r['error']] ?? 'حصلت مشكلة، جرب تاني', err: true);
        return;
      }
      _load();
      if (mounted) toast(context, 'المشترك اتضاف');
    } catch (_) { if (mounted) toast(context, 'حصلت مشكلة', err: true); }
  }

  Future<void> _subAction(String action, Map s) async {
    try {
      if (action == 'toggle') {
        await sb.from('video_students').update({'active': s['active'] != true}).eq('id', s['id']);
      } else if (action == 'delete') {
        final yes = await showDialog<bool>(context: context, builder: (ctx) => Directionality(textDirection: TextDirection.rtl, child: AlertDialog(title: const Text('حذف المشترك ده؟'), actions: [TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('لا')), TextButton(onPressed: () => Navigator.pop(ctx, true), child: const Text('احذف'))])));
        if (yes != true) return;
        await sb.from('video_students').delete().eq('id', s['id']);
      } else if (action == 'reset') {
        final pass = TextEditingController();
        final ok = await showDialog<bool>(
          context: context,
          builder: (ctx) => Directionality(
            textDirection: TextDirection.rtl,
            child: AlertDialog(
              title: const Text('باسورد جديد'),
              content: PwField(controller: pass, label: 'كلمة السر الجديدة'),
              actions: [
                TextButton(onPressed: () => Navigator.pop(ctx, false), child: const Text('إلغاء')),
                ElevatedButton(onPressed: () => Navigator.pop(ctx, true), child: const Text('حفظ')),
              ],
            ),
          ),
        );
        if (ok != true || pass.text.length < 8) { if (mounted) toast(context, 'كلمة السر لازم تكون 8 حروف على الأقل.', err: true); return; }
        final r = await sb.rpc('reset_video_student_password', params: {'p_student': s['id'], 'p_password': pass.text});
        if (r['ok'] != true) { if (mounted) toast(context, 'حصلت مشكلة', err: true); return; }
        if (mounted) toast(context, 'الباسورد اتغير');
      }
      _load();
    } catch (_) { if (mounted) toast(context, 'حصلت مشكلة', err: true); }
  }
